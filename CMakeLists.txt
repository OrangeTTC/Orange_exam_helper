cmake_minimum_required(VERSION 3.21)
project(OrangeStudy VERSION 1.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Sql)
add_library(orange_tinyxml2 STATIC third_party/tinyxml2/tinyxml2.cpp)
target_include_directories(orange_tinyxml2 SYSTEM PUBLIC third_party/tinyxml2)
set_target_properties(orange_tinyxml2 PROPERTIES AUTOMOC OFF)
add_subdirectory(third_party/MicroTeX)

add_library(orange_core src/core/model.cpp src/core/repository.cpp src/core/study_timer.cpp
    src/core/chapter_catalog.cpp src/core/study_analysis.cpp)
target_include_directories(orange_core PUBLIC include)
target_link_libraries(orange_core PUBLIC Qt6::Core Qt6::Sql)
add_library(orange_gui src/gui/main_window.cpp src/gui/question_dialog.cpp src/gui/math_browser.cpp
    src/gui/reading_preferences.cpp src/gui/chapter_picker.cpp src/gui/analysis_page.cpp
    include/orange/reading_preferences.hpp)
target_link_libraries(orange_gui PUBLIC orange_core Qt6::Widgets PRIVATE orange_microtex)
add_executable(OrangeStudy WIN32 src/gui/main.cpp)
target_link_libraries(OrangeStudy PRIVATE orange_gui)
add_executable(orange-cli src/cli/main.cpp)
target_link_libraries(orange-cli PRIVATE orange_core)
foreach(target orange_core orange_gui OrangeStudy orange-cli)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(orange_tests tests/core_tests.cpp)
    target_link_libraries(orange_tests PRIVATE orange_core Qt6::Test)
    add_test(NAME core COMMAND orange_tests -o -,txt)
    add_executable(orange_gui_tests tests/gui_tests.cpp)
    target_link_libraries(orange_gui_tests PRIVATE orange_gui Qt6::Test)
    add_test(NAME gui COMMAND orange_gui_tests -o -,txt)
    add_executable(orange_math_tests tests/math_tests.cpp)
    target_link_libraries(orange_math_tests PRIVATE orange_gui Qt6::Test)
    add_test(NAME math COMMAND orange_math_tests -o -,txt)
    set_tests_properties(gui math PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    set_tests_properties(core gui math PROPERTIES TIMEOUT 30)
endif()
install(TARGETS OrangeStudy orange-cli RUNTIME DESTINATION bin BUNDLE DESTINATION .)
install(FILES LICENSE NOTICE README.md DESTINATION .)
